## Reproduce the lifecycle check without bind

Run from the assigned worktree. This script uses no real sockets, auth or database:

```sh
node <<'JS'
const assert = require('node:assert/strict');
const { EventEmitter } = require('node:events');
const request = require('supertest');
const { Request } = require('superagent');
const originalEnd = Request.prototype.end;
Request.prototype.end = function (callback) {
  process.nextTick(() => callback(null, { status: 401 }));
  return this;
};
class LifecycleServer extends EventEmitter {
  constructor() { super(); this.addr = null; this._handle = null; this.starts = 0; this.stops = 0; }
  address() { return this.addr; }
  listen(port, host) {
    assert.equal(port, 0); assert.equal(host, '127.0.0.1'); this.starts++;
    process.nextTick(() => { this.addr = { address: host, port: 12345 }; this._handle = {}; this.emit('listening'); });
    return this;
  }
  close(callback) {
    this.stops++; this.addr = null; this._handle = null;
    process.nextTick(() => { this.emit('close'); callback(); });
    return this;
  }
}
(async () => {
  try {
    const before = new LifecycleServer(); const beforeHttp = request(before);
    const create = beforeHttp.post('/quotes'); const price = beforeHttp.post('/quotes/q/price-items');
    assert.equal((await create.send({})).status, 401);
    await assert.rejects(async () => await price.send({}), { name: 'TypeError', message: "Cannot read properties of null (reading 'address')" });
    assert.equal(before.starts, 1); assert.equal(before.stops, 1);
    const after = new LifecycleServer(); const afterHttp = request(after);
    assert.equal((await afterHttp.post('/quotes').send({})).status, 401);
    assert.equal((await afterHttp.post('/quotes/q/price-items').send({})).status, 401);
    assert.equal(after.starts, 2); assert.equal(after.stops, 2);
    console.log('PASS actual Supertest lifecycle, no sockets; persisted HTTP/RBAC NOT_TESTED.');
  } finally { Request.prototype.end = originalEnd; }
})().catch(error => { console.error(error); process.exitCode = 1; });
JS
```
